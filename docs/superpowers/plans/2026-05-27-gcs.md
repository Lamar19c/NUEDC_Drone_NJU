def connect_mavlink(params: dict) -> str:
    """params: {type, port?, baud?, host?, port?}  returns connection_string"""
    global mav
    disconnect_mavlink()

    conn_type = params.get("type", "sitl")

    if conn_type == "serial":
        port = params.get("port", "/dev/ttyUSB0")
        baud = params.get("baud", 57600)
        conn_str = f"{port}"
        master = mavutil.mavlink_connection(conn_str, baud=baud)
    elif conn_type == "udp":
        host = params.get("host", "192.168.1.100")
        port = params.get("port", 14550)
        conn_str = f"udp:{host}:{port}"
        master = mavutil.mavlink_connection(conn_str)
    else:  # sitl
        conn_str = "udp:127.0.0.1:14550"
        master = mavutil.mavlink_connection(conn_str)

    master.wait_heartbeat()
    mav = master

    # Start telemetry reader thread
    t = threading.Thread(target=_mavlink_reader, daemon=True)
    t.start()

    print(f"[MAVLink] Connected: {conn_str}, sys={master.target_system}, comp={master.target_component}")
    return conn_str


def disconnect_mavlink():
    global mav
    if mav:
        try:
            mav.close()
        except Exception:
            pass
        mav = None
        print("[MAVLink] Disconnected")
```

def _mavlink_reader():
    """Background thread: reads MAVLink messages, updates telemetry_data dict."""
    global telemetry_data, mission_seq, home_ned

    while mav:
        try:
            msg = mav.recv_match(blocking=True, timeout=1)
            if msg is None:
                continue
        except Exception:
            continue

        msg_type = msg.get_type()

        with telemetry_lock:
            if msg_type == "HEARTBEAT":
                telemetry_data["mode"] = mav.flightmode
                telemetry_data["armed"] = (msg.base_mode & 0x80) != 0

            elif msg_type == "GLOBAL_POSITION_INT":
                lat = msg.lat / 1e7
                lon = msg.lon / 1e7
                rel_alt = msg.relative_alt / 1000.0  # mm → m
                heading = msg.hdg / 100.0 if msg.hdg < 36000 else 0
                telemetry_data["lat"] = lat
                telemetry_data["lon"] = lon
                telemetry_data["alt"] = rel_alt
                telemetry_data["heading"] = heading

                # NED position from GPS deltas (approximate)
                # lat deg → m: 1 deg lat ≈ 111320 m, 1 deg lon ≈ 111320 * cos(lat)
                if "home_lat" not in telemetry_data:
                    telemetry_data["home_lat"] = lat
                    telemetry_data["home_lon"] = lon
                dlat = lat - telemetry_data["home_lat"]
                dlon = lon - telemetry_data["home_lon"]
                cos_lat = math.cos(math.radians(telemetry_data["home_lat"]))
                telemetry_data["ned_y"] = dlat * 111320.0
                telemetry_data["ned_x"] = dlon * 111320.0 * cos_lat
                telemetry_data["ned_z"] = -rel_alt

            elif msg_type == "LOCAL_POSITION_NED":
                telemetry_data["ned_x"] = msg.x
                telemetry_data["ned_y"] = msg.y
                telemetry_data["ned_z"] = msg.z
                telemetry_data["vx"] = msg.vx
                telemetry_data["vy"] = msg.vy
                telemetry_data["vz"] = msg.vz

            elif msg_type == "SYS_STATUS":
                telemetry_data["battery_voltage"] = msg.voltage_battery / 1000.0 if msg.voltage_battery < 65535 else 0
                telemetry_data["battery_percent"] = msg.battery_remaining if msg.battery_remaining < 101 else 0

            elif msg_type == "GPS_RAW_INT":
                telemetry_data["satellites"] = msg.satellites_visible
                telemetry_data["hdop"] = msg.eph / 100.0 if msg.eph < 65535 else 99

            elif msg_type == "MISSION_CURRENT":
                mission_seq = msg.seq
                telemetry_data["mission_seq"] = msg.seq
                telemetry_data["mission_total"] = telemetry_data.get("mission_total", 0)
```

def do_set_mode(mode_name: str) -> bool:
    if not mav:
        return False
    if mode_name not in mav.mode_mapping():
        return False
    mode_id = mav.mode_mapping()[mode_name]
    mav.mav.set_mode_send(mav.target_system,
                          mavutil.mavlink.MAV_MODE_FLAG_CUSTOM_MODE_ENABLED,
                          mode_id)
    return True


def do_arm() -> bool:
    if not mav:
        return False
    mav.mav.command_long_send(mav.target_system, mav.target_component,
                              mavutil.mavlink.MAV_CMD_COMPONENT_ARM_DISARM,
                              0, 1, 0, 0, 0, 0, 0, 0)
    return True


def do_disarm() -> bool:
    if not mav:
        return False
    mav.mav.command_long_send(mav.target_system, mav.target_component,
                              mavutil.mavlink.MAV_CMD_COMPONENT_ARM_DISARM,
                              0, 0, 0, 0, 0, 0, 0, 0)
    return True


def do_takeoff(alt_m: float) -> bool:
    if not mav:
        return False
    mav.mav.command_long_send(mav.target_system, mav.target_component,
                              mavutil.mavlink.MAV_CMD_NAV_TAKEOFF,
                              0, 0, 0, 0, 0, 0, 0, alt_m)
    return True


def do_land() -> bool:
    if not mav:
        return False
    set_mode("LAND")
    return True


def do_goto(north: float, east: float, down: float) -> bool:
    if not mav:
        return False
    type_mask = 0b0000111111111000  # position only
    mav.mav.set_position_target_local_ned_send(
        0, mav.target_system, mav.target_component,
        mavutil.mavlink.MAV_FRAME_LOCAL_NED,
        type_mask, north, east, down, 0, 0, 0, 0, 0, 0, 0, 0)
    return True


def do_mission_upload(waypoints: list) -> bool:
    """waypoints: [{x, y, z, delay, yaw}, ...] in NED meters"""
    if not mav:
        return False
    count = len(waypoints)
    # Send count first
    mav.mav.mission_count_send(mav.target_system, mav.target_component, count, 0)
    for i, wp in enumerate(waypoints):
        lat, lon, alt = _ned_to_global(wp["x"], wp["y"], wp["z"])
        mav.mav.mission_item_send(
            mav.target_system, mav.target_component,
            i, mavutil.mavlink.MAV_FRAME_GLOBAL_RELATIVE_ALT,
            mavutil.mavlink.MAV_CMD_NAV_WAYPOINT,
            0, 1,  # current, autocontinue
            wp.get("delay", 0),  # param1: hold time (s)
            0, 0,  # param2-3: acceptance radius, pass radius
            wp.get("yaw", 0),  # param4: yaw (deg)
            lat, lon, alt  # x, y, z (lat, lon, alt)
        )
    with telemetry_lock:
        telemetry_data["mission_total"] = count
    return True


def do_mission_start() -> bool:
    if not mav:
        return False
    mav.mav.command_long_send(mav.target_system, mav.target_component,
                              mavutil.mavlink.MAV_CMD_MISSION_START,
                              0, 0, 0, 0, 0, 0, 0, 0)
    return True


def do_mission_pause() -> bool:
    if not mav:
        return False
    # Pause by switching to GUIDED, preserving current position
    mav.mav.command_long_send(mav.target_system, mav.target_component,
                              mavutil.mavlink.MAV_CMD_DO_PAUSE_CONTINUE,
                              0, 0, 0, 0, 0, 0, 0, 0)
    return True


def _ned_to_global(north: float, east: float, down: float) -> tuple:
    """Convert NED meters to lat(E7), lon(E7), alt(m) using stored home."""
    with telemetry_lock:
        home_lat = telemetry_data.get("home_lat", 0)
        home_lon = telemetry_data.get("home_lon", 0)
    lat_rad = math.radians(home_lat)
    lat = int((home_lat + north / 111320.0) * 1e7)
    lon = int((home_lon + east / (111320.0 * math.cos(lat_rad))) * 1e7)
    alt = -down
    return lat, lon, alt
```

async def ws_handler(websocket):
    """Handle one WebSocket client."""
    ws_clients.add(websocket)
    addr = websocket.remote_address
    print(f"[WS] Client connected: {addr}")
    try:
        async for message in websocket:
            try:
                data = json.loads(message)
            except json.JSONDecodeError:
                await websocket.send(json.dumps({"type": "status", "msg": "Invalid JSON", "level": "error"}))
                continue

            cmd = data.get("cmd", "")
            response = {"type": "status", "msg": "ok"}

            if cmd == "connect":
                conn_str = connect_mavlink(data)
                response = {"type": "connected", "system": mav.target_system if mav else 0,
                            "component": mav.target_component if mav else 0,
                            "connection": conn_str}
            elif cmd == "disconnect":
                disconnect_mavlink()
                response = {"type": "disconnected"}
            elif cmd == "set_mode":
                ok = do_set_mode(data.get("mode", "GUIDED"))
                response = {"type": "status", "msg": f"mode={'ok' if ok else 'fail'}"}
            elif cmd == "arm":
                ok = do_arm()
                response = {"type": "status", "msg": f"arm={'ok' if ok else 'fail'}"}
            elif cmd == "disarm":
                ok = do_disarm()
                response = {"type": "status", "msg": f"disarm={'ok' if ok else 'fail'}"}
            elif cmd == "takeoff":
                ok = do_takeoff(data.get("alt", 3.0))
                response = {"type": "status", "msg": f"takeoff={'ok' if ok else 'fail'}"}
            elif cmd == "land":
                ok = do_land()
                response = {"type": "status", "msg": f"land={'ok' if ok else 'fail'}"}
            elif cmd == "goto":
                ok = do_goto(data.get("north", 0), data.get("east", 0), data.get("down", 0))
                response = {"type": "status", "msg": f"goto={'ok' if ok else 'fail'}"}
            elif cmd == "upload_mission":
                wps = data.get("waypoints", [])
                ok = do_mission_upload(wps)
                response = {"type": "status", "msg": f"mission_upload={'ok' if ok else 'fail'} ({len(wps)} wpts)"}
            elif cmd == "mission_start":
                ok = do_mission_start()
                response = {"type": "status", "msg": f"mission_start={'ok' if ok else 'fail'}"}
            elif cmd == "mission_pause":
                ok = do_mission_pause()
                response = {"type": "status", "msg": f"mission_pause={'ok' if ok else 'fail'}"}
            elif cmd == "set_param":
                global indoor_mode
                if "indoor_mode" in data:
                    indoor_mode = data["indoor_mode"]
                response = {"type": "status", "msg": f"indoor_mode={indoor_mode}"}
            else:
                response = {"type": "status", "msg": f"unknown cmd: {cmd}", "level": "warn"}

            await websocket.send(json.dumps(response))

    except websockets.exceptions.ConnectionClosed:
        pass
    finally:
        ws_clients.discard(websocket)
        print(f"[WS] Client disconnected: {addr}")


async def telemetry_broadcast():
    """Send telemetry JSON to all connected WS clients at ~4 Hz."""
    while True:
        await asyncio.sleep(0.25)
        if not ws_clients:
            continue
        with telemetry_lock:
            t = dict(telemetry_data)
        t["type"] = "telemetry"
        t["mission_seq"] = mission_seq
        payload = json.dumps(t)
        stale = set()
        for ws in ws_clients:
            try:
                await ws.send(payload)
            except websockets.exceptions.ConnectionClosed:
                stale.add(ws)
        ws_clients.difference_update(stale)


async def run_server(ws_port: int):
    """Start WebSocket server + telemetry loop."""
    async with serve(ws_handler, "0.0.0.0", ws_port):
        print(f"[WS] Server listening on ws://0.0.0.0:{ws_port}")
        await telemetry_broadcast()
```

function executeNextWaypoint(wpts) {
  if (!stepByStepActive || stepByStepIndex >= wpts.length) {
    stopStepByStep();
    logStatus('逐步执行完成', 'info');
    return;
  }
  const wp = wpts[stepByStepIndex];
  wsSend('goto', { north: wp.x, east: wp.y, down: wp.z });
  logStatus(`→ WP${stepByStepIndex + 1} (${wp.x}, ${wp.y}, ${wp.z}) 悬停${wp.delay}s`, 'info');

  // Wait for reach + hover time before next
  const waitMs = Math.max(wp.delay || 3, 1) * 1000 + 5000; // delay + 5s transit estimate
  stepByStepTimer = setTimeout(() => {
    stepByStepIndex++;
    const currentWpts = getWaypointsFromTable();
    executeNextWaypoint(currentWpts);
  }, waitMs);
}

function stopStepByStep() {
  stepByStepActive = false;
  if (stepByStepTimer) { clearTimeout(stepByStepTimer); stepByStepTimer = null; }
